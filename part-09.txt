    function handleLeadClick(event) {
      const personButton = event.target.closest("[data-person]");
      const editButton = event.target.closest("[data-edit]");
      const duplicateButton = event.target.closest("[data-duplicate]");
      if (personButton) openPersonModal(personButton.dataset.person);
      if (editButton) openSessionDrawer(state.sessions.find(s => s.id === editButton.dataset.edit));
      if (duplicateButton) duplicateSession(duplicateButton.dataset.duplicate);
    }

    function openSessionDrawer(session = null) {
      editingId = session ? session.id : null;
      const data = session || sessionDefaults();
      els.drawerTitle.textContent = session ? "Edit Session" : "Add Session";
      els.deleteSession.classList.toggle("hidden", !session);
      setForm(data);
      els.duplicatePrompt.classList.remove("show");
      renderChipSelection();
      els.sessionDrawer.classList.add("show");
      els.sessionDrawer.setAttribute("aria-hidden", "false");
      setTimeout(() => els.name.focus(), 80);
    }

    function closeSessionDrawer() {
      els.sessionDrawer.classList.remove("show");
      els.sessionDrawer.setAttribute("aria-hidden", "true");
      editingId = null;
    }

      els.sessionId.value = session.id || "";

      els.stage.value = session.stage || "Message";
      els.priority.value = session.priority || "Med";

      const existing = state.sessions.find(s => s.id === els.sessionId.value);
      const now = new Date().toISOString();
      return {
        id: els.sessionId.value || uid(),
        date: els.date.value || TODAY,
        name: els.name.value.trim(),
        company: els.company.value.trim(),

        priority: els.priority.value,
        reason: els.reason.value.trim(),

        leadSource: els.leadSource.value,

        nextAction: els.nextAction.value.trim(),

        updatedAt: now
      };
    }

    async function saveSessionFromForm(event) {
      event.preventDefault();
      const session = getFormSession();
      if (!session.name) {
        toast("Add a name first");
        els.name.focus();
        return;
      }
      saveEmergencyDraft(session);
      if (currentUser) {
        const saved = await saveCloudSessionRecord(session);
        if (!saved) return;
      } else {
        const index = state.sessions.findIndex(s => s.id === session.id);
        if (index >= 0) state.sessions[index] = session;
        else state.sessions.unshift(session);
        saveState();
      }
      closeSessionDrawer();
      render();
      toast("Session saved");
    }

    async function deleteCurrentSession() {
      if (!editingId) return;
      const session = state.sessions.find(s => s.id === editingId);
      if (!confirm(`Delete this session for ${session?.name || "this contact"}?`)) return;
      if (currentUser) {
        const deleted = await deleteCloudSession(editingId);
        if (!deleted) return;
      }
      state.sessions = state.sessions.filter(s => s.id !== editingId);
      if (!currentUser) saveState();
      closeSessionDrawer();
      render();
      toast("Session deleted");
    }

    function duplicateSession(id) {
      const original = state.sessions.find(s => s.id === id);
      if (!original) return;
      const now = new Date().toISOString();
      const copy = { ...original, id: uid(), date: TODAY, lastContacted: TODAY, followUp: "", stage: "Message", notes: "", nextAction: "", createdAt: now, updatedAt: now };
      openSessionDrawer(copy);
      editingId = null;
      els.drawerTitle.textContent = "Add New Session";
      checkDuplicateName();
    }
